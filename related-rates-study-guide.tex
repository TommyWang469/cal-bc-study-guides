\section*{Answer Key}
\textit{Signed rates are shown where helpful. A negative rate means the quantity is decreasing.}
\begin{description}[leftmargin=2.8em,labelwidth=2em,itemsep=13pt]
\ansitem{1} \textbf{(B)} $\displaystyle\frac{dx}{dt}=-\frac65$.
\ansitem{2} $\displaystyle\frac{dx}{dt}=\frac94$.
\ansitem{3} The area increases at $1800\pi$ ft$^2$/min.
\ansitem{4} \textbf{(B)} $\displaystyle\frac{dA}{dt}=-0.1C$ cm$^2$/sec, when $C$ is measured in centimeters.
\ansitem{5} $\displaystyle\frac{dV_{\rm ice}}{dt}=-7442\pi$ in.$^3$/hr, equivalently $-\dfrac{3721\pi}{864}$ ft$^3$/hr.
\ansitem{6} $\displaystyle\frac{dr}{dt}=\frac{20}{9\pi}$ ft/min.
\ansitem{7} The distance increases at $\displaystyle\frac{66}{\sqrt{29}}\approx12.256$ ft/sec.
\ansitem{8} $\displaystyle\frac{dd}{dt}=-\frac{3750}{\sqrt{429}}\approx-181.052$ mi/hr; the distance is decreasing.
\ansitem{9} \textbf{(E)} $RQ=32$ ft.
\ansitem{10} The boat approaches the dock at $\displaystyle\frac{2\sqrt{674}}{25}\approx2.077$ ft/sec. Its horizontal distance from the dock has the negative of this rate.
\end{description}

\section*{Answer Key \textnormal{(continued)}}
\begin{description}[leftmargin=2.8em,labelwidth=2em,itemsep=11pt]
\ansitem{11} The shadow shortens at $1$ ft/sec: $s'=-1$ ft/sec.
\par\smallskip Let $H$ be the lamp height and $s$ the shadow length. Similar triangles give
\[\frac{H}{12+s}=\frac6s \quad\Longrightarrow\quad (H-6)s=72.\]
Differentiate: $H's+(H-6)s'=0$. When $H=18$, $s=6$ and $H'=2$, so $12+12s'=0$ and $s'=-1$ ft/sec.
\ansitem{12} The shadow height decreases at $\dfrac{15}{16}$ m/sec: $y'=-\dfrac{15}{16}$ m/sec.
\par\smallskip Let $x$ be the person's distance from the spotlight and $y$ the shadow height on the wall. Similar triangles give
\[\frac{y}{20}=\frac2x \quad\Longrightarrow\quad xy=40.\]
Differentiate: $x'y+xy'=0$. At $x=8$, $y=5$ and $x'=\frac32$, so $\frac{15}{2}+8y'=0$ and $y'=-\frac{15}{16}$ m/sec.
\ansitem{13} Sand leaks into the pile at $600\pi$ in.$^3$/min.
\ansitem{14} The level decreases at $\displaystyle\frac{27}{25\pi}$ in./hr; $h'=-\dfrac{27}{25\pi}$ in./hr.
\ansitem{15} The water level rises at $\displaystyle\frac{15\sqrt3}{32}\approx0.812$ ft/min.
\ansitem{16} The water level rises at approximately $0.042853$ ft/min.
\ansitem{17} The airplane's speed is $\displaystyle\frac{2000\pi}{27}\approx232.711$ ft/sec.
\ansitem{18} The missile's velocity is $\displaystyle\frac{2\pi}{27}$ mi/sec upward, equivalently $\dfrac{3520\pi}{9}\approx1228.712$ ft/sec upward.
\ansitem{19} The volume increases at $5$ in.$^3$/min.
\ansitem{20} The total resistance increases at $\displaystyle\frac{11}{1600}=0.006875$ ohm/sec.
\end{description}
